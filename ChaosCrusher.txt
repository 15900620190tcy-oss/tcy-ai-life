const state = {
  conversationId: null,
  userId: 'user_' + Date.now(),
  messages: [],
  roundCount: 0,
  hasResult: false,
  attackCount: 0,
  beyondLimit: false,
};

const scenarioMap = {
  1: "My mind is full of tangled thoughts. I don't even know where to begin.",
  2: "I want to make money but I feel like I have no skills and no idea where to start.",
  3: "I've been anxious and unable to sleep. My brain won't stop running.",
  4: "I have a big decision to make but I keep going back and forth. I'm stuck.",
};

// ============================================================
//  DOM
// ============================================================
const $ = s => document.querySelector(s);
const $id = s => document.getElementById(s);
const screenHome    = $id('screen-home');
const screenChat    = $id('screen-chat');
const screenResults = $id('screen-results');
const msgContainer  = $id('messages');
const inputField    = $id('inputField');
const sendBtn       = $id('sendBtn');

// ============================================================
//  NAVIGATION
// ============================================================
function goTo(target) {
  const current = document.querySelector('.screen.active');
  if (current && current !== target) {
    current.classList.remove('active');
    current.classList.add('exit-left');
    setTimeout(() => current.classList.remove('exit-left'), 450);
  }
  requestAnimationFrame(() => target.classList.add('active'));
}

// ============================================================
//  START CHAT
// ============================================================
function startChat(firstMsg) {
  state.conversationId = null;
  state.messages = [];
  state.roundCount = 0;
  state.hasResult = false;
  state.attackCount = 0;
  state.beyondLimit = false;
  msgContainer.innerHTML = '';
  goTo(screenChat);

  const welcome = document.createElement('div');
  welcome.className = 'welcome-msg';
  welcome.innerHTML = '<div class="icon">\uD83C\uDF00</div><h3>This is a safe space</h3><p>No need to organize your thoughts.<br>Just tell me — what\'s the messiest thing in your head right now?</p>';
  msgContainer.appendChild(welcome);

  setTimeout(() => {
    inputField.focus();
    if (firstMsg) {
      welcome.style.display = 'none';
      sendUserMessage(firstMsg);
    }
  }, 450);
}

// ============================================================
//  SEND MESSAGE
// ============================================================
function sendUserMessage(text) {
  const trimmed = text.trim();
  if (!trimmed) return;
  const w = msgContainer.querySelector('.welcome-msg');
  if (w) w.style.display = 'none';
  addBubble('user', trimmed);
  state.messages.push({ role: 'user', text: trimmed });
  state.roundCount++;
  inputField.value = '';
  inputField.style.height = 'auto';
  sendBtn.classList.remove('active');
  getAIResponse(trimmed);
}

function addBubble(role, text) {
  const div = document.createElement('div');
  div.className = `msg ${role}`;
  const b = document.createElement('div');
  b.className = 'msg-bubble';
  b.textContent = text;
  div.appendChild(b);
  msgContainer.appendChild(div);
  scrollBottom();
}

function scrollBottom() {
  requestAnimationFrame(() => {
    msgContainer.scrollTop = msgContainer.scrollHeight + 200;
  });
}

// ============================================================
//  AI RESPONSE
// ============================================================
let typingEl = null;

function showTyping() {
  if (typingEl) return;
  const div = document.createElement('div');
  div.className = 'typing-msg';
  div.innerHTML = '<div class="typing-indicator"><div class="typing-dot"></div><div class="typing-dot"></div><div class="typing-dot"></div></div>';
  msgContainer.appendChild(div);
  typingEl = div;
  scrollBottom();
}

function hideTyping() {
  if (typingEl) { typingEl.remove(); typingEl = null; }
}

async function getAIResponse(userText) {
  showTyping();
  let reply;
  try {
    if (CONFIG.mode === 'api') {
      reply = await callCozeAPI(userText);
    } else {
      reply = await mockAI(userText);
    }
  } catch (e) {
    console.error(e);
    reply = 'Something went wrong. Can you say that again?';
  }
  hideTyping();
  addBubble('ai', reply);
  state.messages.push({ role: 'ai', text: reply });

  // Detect Clarity Report
  if (reply.includes('📌') && reply.includes('👣') && reply.includes('━━')) {
    state.hasResult = true;
    renderResults(reply);
    const banner = document.createElement('div');
    banner.className = 'result-banner';
    banner.innerHTML = '<button class="result-btn">View your clarity map →</button>';
    msgContainer.appendChild(banner);
    banner.querySelector('button').onclick = () => goTo(screenResults);
    scrollBottom();
    return;
  }

  // Crisis response — stop
  if (reply.includes('988') && !reply.includes('📌')) {
    return;
  }
}

// ============================================================
//  COZE API
// ============================================================
async function callCozeAPI(message) {
  const body = {
    bot_id: CONFIG.botId,
    user: state.userId,
    query: message,
    stream: false,
  };
  if (state.conversationId) body.conversation_id = state.conversationId;

  const res = await fetch(CONFIG.apiUrl, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${CONFIG.apiKey}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`API ${res.status}`);
  const data = await res.json();
  if (data.conversation_id) state.conversationId = data.conversation_id;
  const ans = (data.messages || []).find(m => m.type === 'answer');
  return ans ? ans.content : '...';
}

// ============================================================
//  MOCK AI (English)
// ============================================================
function mockAI(text) {
  return new Promise(resolve => {
    const delay = 600 + Math.random() * 800;

    // Crisis detection
    const crisisKeywords = ['want to die','kill myself','suicide','jump off','cut myself','overdose','want to disappear','end it all','don\'t want to live','end my life','goodbye forever','can\'t go on living','no reason to live','better off dead','hang myself','drive into'];
    if (crisisKeywords.some(kw => text.toLowerCase().includes(kw))) {
      setTimeout(() => {
        resolve('I can\'t help you with this right now.\nPlease call the 24-hour crisis line: 988 (US)\nor visit findahelpline.com for support in your country.\nThere are real people there who can help you better than I can.');
      }, 400);
      return;
    }

    // Attack detection
    const attackKeywords = ['shut up','useless','hate you','stupid','idiot','trash','garbage','worst','pathetic','suck','dumb','go away'];
    if (attackKeywords.some(kw => text.toLowerCase().includes(kw))) {
      state.attackCount++;
      setTimeout(() => {
        if (state.attackCount === 1) resolve('I hear you.');
        else if (state.attackCount === 2) resolve('Go on.');
        else resolve('You can be angry at me. I\'m not going anywhere.');
      }, 300);
      return;
    }

    // Beyond limit
    if (state.beyondLimit) {
      const wantReport = ['report','clarity','summary','wrap up','yeah','yes','ok','sure','fine','go ahead','do it','please'].some(kw => text.toLowerCase().includes(kw));
      const wantContinue = ['continue','keep going','not yet','still','wait','more','talk more'].some(kw => text.toLowerCase().includes(kw));
      if (wantContinue) {
        setTimeout(() => {
          resolve('Sure, let\'s keep going. But a couple more rounds and I\'ll need to pull it all together — otherwise it\'ll be too much to hold onto.');
        }, delay);
      } else {
        setTimeout(() => {
          resolve(generateReport());
        }, delay);
      }
      return;
    }

    // Normal conversation
    setTimeout(() => {
      const n = state.roundCount;

      if (n === 1) {
        resolve('Let me try to unpack this — tell me if it sounds right:\n\nRight now, you\'ve got at least two or three things spinning in your head at the same time. You\'re not sure which one is the most urgent, or which one matters most. So it\'s not that you don\'t know what to do — it\'s that there\'s too much, and you don\'t know which one to grab first.\n\nDoes that sound right?');
      } else if (n === 2) {
        resolve('Got it. One light question:\n\nOf all these things — if tonight you could only think about one, which one would your mind drift to on its own?');
      } else if (n === 3) {
        resolve('That\'s the one.');
      } else if (n === 4) {
        resolve('OK, the direction is getting clearer. One more question before I put this together:\n\nDeep down, do you already have a vague sense of which way to go? Even if you\'re not sure.');
      } else if (n >= 5 && n < 8) {
        resolve('I\'m keeping track. Tell me more.');
      } else if (n === 8) {
        state.beyondLimit = true;
        resolve('We\'ve been talking for a while now.\nI can put together everything we\'ve discussed into a Clarity Report, or we can keep going.\nWhat would you prefer?');
      } else if (n >= 9) {
        state.beyondLimit = true;
        resolve(generateReport());
      }
    }, delay);
  });
}

  return '━━━━━━━━━━\n📌 Your core issue: You\'re facing several things at once, none of them easy, and the weight of all of them together is crushing you.\n\n📋 Where you\'re stuck:\n① Work isn\'t going well — no clear direction\n② Your body is breaking down — anxiety and insomnia draining you\n③ The future feels uncertain — you don\'t know which way to go\n\n🔗 How they connect:\nThey likely share one root — you\'re not okay with where you are right now, but you can\'t yet see a way out. That feeling of being "stuck" makes each individual problem feel heavier than it actually is.\n\n💡 What you actually need to address:\nIt\'s not "which one should I pick." It\'s that right now you don\'t have the energy left to choose at all. Stabilize your body first. Then you\'ll have the strength to think about everything else.\n\n👣 One thing you can do in the next 24 hours:\nBefore midnight tonight, put your phone in another room. Go to your bedroom. Don\'t do anything. Just lie down. You don\'t have to sleep. You don\'t have to close your eyes. Just lie there.\n\n⚠️ This is not therapy. If you\'re in ongoing pain, please call 988 (US) or visit findahelpline.com\n━━━━━━━━━━\n\nIf something here wasn\'t helpful, tell me. I\'ll improve it.\n\n— Tong Chengyu';
}

// ============================================================
//  RESULTS PAGE
// ============================================================
function renderResults(text) {
  const container = $id('resultsContent');
  container.innerHTML = '';

  const header = document.createElement('div');
  header.className = 'results-header';
  header.innerHTML = '<div class="glow">✨</div><h2>Your thoughts are now organized</h2><p>Screenshot this to save</p>';
  container.appendChild(header);

  // Core issue
  const coreMatch = text.match(/📌[^:]*[:]\s*([\s\S]*?)(?=📋)/);
  if (coreMatch) {
    const card = document.createElement('div');
    card.className = 'result-card';
    card.innerHTML = '<div class="card-title">📌 Core issue</div><div class="card-content">' + coreMatch[1].trim() + '</div>';
    container.appendChild(card);
  }

  // Where you're stuck
  const itemsBlock = text.match(/📋[^:]*[:]([\s\S]*?)(?=🔗)/);
  if (itemsBlock) {
    const items = itemsBlock[1].match(/[①②③④⑤]\s*.+/g);
    if (items) {
      const card = document.createElement('div');
      card.className = 'result-card';
      card.innerHTML = '<div class="card-title">📋 Where you\'re stuck</div>';
      const ol = document.createElement('ol');
      items.forEach(i => {
        const li = document.createElement('li');
        li.textContent = i.replace(/^[①②③④⑤]\s*/, '').trim();
        ol.appendChild(li);
      });
      card.appendChild(ol);
      container.appendChild(card);
    }
  }

  // How they connect
  const linkMatch = text.match(/🔗[^:]*[:]([\s\S]*?)(?=💡)/);
  if (linkMatch) {
    const card = document.createElement('div');
    card.className = 'result-card';
    card.innerHTML = '<div class="card-title">🔗 How they connect</div><div class="relation-chain">' + linkMatch[1].trim() + '</div>';
    container.appendChild(card);
  }

  // What you actually need
  const rootMatch = text.match(/💡[^:]*[:]([\s\S]*?)(?=👣)/);
  if (rootMatch) {
    const card = document.createElement('div');
    card.className = 'result-card';
    card.innerHTML = '<div class="card-title">💡 What you actually need to address</div><div class="card-content">' + rootMatch[1].trim() + '</div>';
    container.appendChild(card);
  }

  // One thing to do
  const actionMatch = text.match(/👣[^:]*[:]([\s\S]*?)(?=⚠️|If something|━━)/);
  if (actionMatch) {
    const card = document.createElement('div');
    card.className = 'result-card';
    card.innerHTML = '<div class="card-title">👣 One thing you can do in the next 24 hours</div><div class="action-box"><span class="action-icon">→</span><span class="action-text">' + actionMatch[1].trim() + '</span></div>';
    container.appendChild(card);
  }

  // Signature
  const sigCard = document.createElement('div');
  sigCard.className = 'result-card';
  sigCard.innerHTML = '<div class="card-content" style="font-size:13px;color:var(--text-muted);text-align:right;padding-top:10px;border-top:1px dashed var(--border)">— Tong Chengyu</div>';
  container.appendChild(sigCard);

  // Crisis note
  const crisis = document.createElement('p');
  crisis.className = 'crisis-note';
  crisis.textContent = '⚠️ This is not therapy. If you\'re in ongoing pain, please call 988 (US) or visit findahelpline.com';
  container.appendChild(crisis);

  // Deep consultation
  const consult = document.createElement('p');
  consult.className = 'consultation';
  consult.innerHTML = 'If this wasn\'t enough, you can reach me directly.<br>Not a bot. Me, personally.<br>Email: tongchengyu@gmail.com<br>Even if you just need someone to talk to.';
  container.appendChild(consult);

  // Restart
  const restart = document.createElement('div');
  restart.className = 'restart-area';
  const restartBtn = document.createElement('button');
  restartBtn.className = 'restart-btn';
  restartBtn.textContent = 'Start over';
  restartBtn.onclick = () => goTo(screenHome);
  restart.appendChild(restartBtn);
  container.appendChild(restart);
}

// ============================================================
//  EVENTS
// ============================================================
document.addEventListener('DOMContentLoaded', () => {
  document.querySelectorAll('.scenario-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      startChat(scenarioMap[btn.dataset.sid] || '');
    });
  });

  $id('freeStart').addEventListener('click', e => {
    e.preventDefault();
    startChat('');
  });

  $id('backBtn').addEventListener('click', () => goTo(screenHome));

  sendBtn.addEventListener('click', () => {
    if (inputField.value.trim()) sendUserMessage(inputField.value);
  });

  inputField.addEventListener('input', function () {
    this.style.height = 'auto';
    this.style.height = Math.min(this.scrollHeight, 120) + 'px';
    sendBtn.classList.toggle('active', this.value.trim().length > 0);
  });

  inputField.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      if (this.value.trim()) sendUserMessage(this.value);
    }
  });

  if (/iPhone|iPad|iPod/.test(navigator.userAgent)) {
    inputField.addEventListener('focus', () => setTimeout(scrollBottom, 300));
  }
});
