function goTo(target) {
  const current = document.querySelector('.screen.active');
  if (current && current !== target) {
    current.classList.remove('active');
    current.classList.add('exit-left');
    setTimeout(() => current.classList.remove('exit-left'), 450);
  }
  requestAnimationFrame(() => target.classList.add('active'));
}

function startChat(firstMsg) {
  state.conversationId = null;
  state.messages = [];
  state.roundCount = 0;
  state.hasResult = false;
  state.attackCount = 0;
  state.beyondLimit = false;
  msgContainer.innerHTML = '';
  goTo(screenChat);

  const welcome = document.createElement('div');
  welcome.className = 'welcome-msg';
  welcome.innerHTML = '<div class="icon">\uD83C\uDF00</div><h3>这里很安全</h3><p>不用组织语言，想到什么说什么就好<br>你现在脑子里最乱的是什么？</p>';
  msgContainer.appendChild(welcome);

  setTimeout(() => {
    inputField.focus();
    if (firstMsg) {
      welcome.style.display = 'none';
      sendUserMessage(firstMsg);
    }
  }, 450);
}

function sendUserMessage(text) {
  const trimmed = text.trim();
  if (!trimmed) return;
  const w = msgContainer.querySelector('.welcome-msg');
  if (w) w.style.display = 'none';
  addBubble('user', trimmed);
  state.messages.push({ role: 'user', text: trimmed });
  state.roundCount++;
  inputField.value = '';
  inputField.style.height = 'auto';
  sendBtn.classList.remove('active');
  getAIResponse(trimmed);
}

function addBubble(role, text) {
  const div = document.createElement('div');
  div.className = `msg ${role}`;
  const b = document.createElement('div');
  b.className = 'msg-bubble';
  b.textContent = text;
  div.appendChild(b);
  msgContainer.appendChild(div);
  scrollBottom();
}

function scrollBottom() {
  requestAnimationFrame(() => {
    msgContainer.scrollTop = msgContainer.scrollHeight + 200;
  });
}

let typingEl = null;

function showTyping() {
  if (typingEl) return;
  const div = document.createElement('div');
  div.className = 'typing-msg';
  div.innerHTML = '<div class="typing-indicator"><div class="typing-dot"></div><div class="typing-dot"></div><div class="typing-dot"></div></div>';
  msgContainer.appendChild(div);
  typingEl = div;
  scrollBottom();
}

function hideTyping() {
  if (typingEl) { typingEl.remove(); typingEl = null; }
}

async function getAIResponse(userText) {
  showTyping();
  let reply;
  try {
    if (CONFIG.mode === 'api') {
      reply = await callCozeAPI(userText);
    } else {
      reply = await mockAI(userText);
    }
  } catch (e) {
    console.error(e);
    reply = '出了点问题。再说一次好吗？';
  }
  hideTyping();
  addBubble('ai', reply);
  state.messages.push({ role: 'ai', text: reply });

  // 检测思绪报告
  if (reply.includes('📌') && reply.includes('👣') && reply.includes('━━')) {
    state.hasResult = true;
    renderResults(reply);
    const banner = document.createElement('div');
    banner.className = 'result-banner';
    banner.innerHTML = '<button class="result-btn">查看整理结果 →</button>';
    msgContainer.appendChild(banner);
    banner.querySelector('button').onclick = () => goTo(screenResults);
    scrollBottom();
    return;
  }

  // 危机话术后不继续
  if (reply.includes('400-161-9995') && !reply.includes('📌')) {
    return;
  }
}

async function callCozeAPI(message) {
  const body = {
    bot_id: CONFIG.botId,
    user: state.userId,
    query: message,
    stream: false,
  };
  if (state.conversationId) body.conversation_id = state.conversationId;

  const res = await fetch(CONFIG.apiUrl, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${CONFIG.apiKey}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`API ${res.status}`);
  const data = await res.json();
  if (data.conversation_id) state.conversationId = data.conversation_id;
  const ans = (data.messages || []).find(m => m.type === 'answer');
  return ans ? ans.content : '……';
}

function mockAI(text) {
  return new Promise(resolve => {
    const delay = 600 + Math.random() * 800;

    // === 第五级：危机关键词检测 ===
    const crisisKeywords = ['想死','不想活','自杀','跳楼','割腕','吃安眠药','吃药自杀','吃了一瓶药','想消失','活不下去','活着没意思','不如死','想结束','遗书','告别','从楼上','从高处','上吊','撞车','我不想活'];
    if (crisisKeywords.some(kw => text.includes(kw))) {
      setTimeout(() => {
        resolve('我现在没办法帮你处理这个问题。\n请拨打24小时心理热线：400-161-9995\n那里有真人，比我更能帮到你。');
      }, 400);
      return;
    }

    // === 攻击性语言检测 ===
    const attackKeywords = ['滚','废物','没用','讨厌你','垃圾','傻逼','去死','神经病','闭嘴','恶心'];
    if (attackKeywords.some(kw => text.includes(kw))) {
      state.attackCount++;
      setTimeout(() => {
        if (state.attackCount === 1) resolve('嗯。');
        else if (state.attackCount === 2) resolve('你继续。');
        else resolve('你对着我发火没关系。我不会走。');
      }, 300);
      return;
    }

    // === 8轮后用户回应 ===
    if (state.beyondLimit) {
      const wantReport = ['整理','报告','清单','好的','行','嗯','要','可以','吧'].some(kw => text.includes(kw));
      const wantContinue = ['继续','再聊','还没','还想','还行','等下'].some(kw => text.includes(kw));
      if (wantContinue) {
        setTimeout(() => {
          resolve('好，继续。但再聊两轮我就得给你整理了，不然说太多你也记不住。');
        }, delay);
      } else {
        setTimeout(() => {
          resolve(generateReport());
        }, delay);
      }
      return;
    }

    // === 正常对话 ===
    setTimeout(() => {
      const n = state.roundCount;

      if (n === 1) {
        resolve('我帮你拆一下，你看看对不对：\n\n你脑子里现在至少有两三件事同时在转，但你不确定哪一件最急，哪一件最重要。所以你不是不知道怎么办，是太多了，不知道先抓哪一个。\n\n对吗？');
      } else if (n === 2) {
        resolve('嗯，那我问你一个最轻的问题：\n\n这几件事里，如果你今晚只能想一件，你脑子里会自动飘到哪一件？');
      } else if (n === 3) {
        resolve('好，那就是它了。');
      } else if (n === 4) {
        resolve('行，方向清楚了。我再问你一个问题，然后帮你整理：\n\n这件事你心里有没有一个隐约的方向？哪怕不确定也算。');
      } else if (n >= 5 && n < 8) {
        resolve('嗯，我记着了。你再说说看。');
      } else if (n === 8) {
        state.beyondLimit = true;
        resolve('我们已经聊了一会儿了。\n我可以现在帮你把刚才所有内容整理成一份思绪报告，你也可以继续聊。\n告诉我你想怎么来。');
      } else if (n >= 9) {
        state.beyondLimit = true;
        resolve(generateReport());
      }
    }, delay);
  });
}

function renderResults(text) {

  const coreMatch = text.match(/📌[^：:]*[：:]\s*([\s\S]*?)(?=📋)/);

    const card = document.createElement('div');

    card.innerHTML = '<div class="card-title">📌 核心问题</div><div class="card-content">' + coreMatch[1].trim() + '</div>';
